### Database Schema Changes
```sql
-- Students
ALTER TABLE students ADD COLUMN blood_group VARCHAR(10);
ALTER TABLE students ADD COLUMN medical_notes TEXT;
ALTER TABLE students ADD COLUMN linkedin_url VARCHAR(255);

-- New Tables
CREATE TABLE student_emergency_contacts (
  id INT PRIMARY KEY AUTO_INCREMENT,
  student_id INT,
  name VARCHAR(100),
  relationship VARCHAR(50),
  mobile VARCHAR(15),
  priority INT
);

CREATE TABLE student_notes (
  id INT PRIMARY KEY AUTO_INCREMENT,
  student_id INT,
  note TEXT,
  created_by INT,
  created_at TIMESTAMP
);

CREATE TABLE student_tags (
  id INT PRIMARY KEY AUTO_INCREMENT,
  student_id INT,
  tag_name VARCHAR(50)
);

-- Admissions
ALTER TABLE admissions ADD COLUMN approval_status ENUM('Pending', 'Approved', 'Rejected');
ALTER TABLE admissions ADD COLUMN approved_by INT;
ALTER TABLE admissions ADD COLUMN approved_at TIMESTAMP;

CREATE TABLE admission_checklist (
  id INT PRIMARY KEY AUTO_INCREMENT,
  admission_id INT,
  item_name VARCHAR(100),
  is_completed BOOLEAN,
  completed_at TIMESTAMP
);

-- Documents
ALTER TABLE student_documents ADD COLUMN verification_status ENUM('Pending', 'Verified', 'Rejected');
ALTER TABLE student_documents ADD COLUMN verified_by INT;
ALTER TABLE student_documents ADD COLUMN verified_at TIMESTAMP;
ALTER TABLE student_documents ADD COLUMN expiry_date DATE;
```
